Google Apps Script (GAS) 백엔드 코드

구글 스프레드시트와 연동되는 완벽한 Apps Script 소스코드

🚨 오류 해결: "Exception: Index(이)라는 이름의 HTML 파일을 찾을 수 없습니다."

원인: Google Apps Script의 Code.gs에서 HtmlService.createHtmlOutputFromFile('Index')를 호출했으나, Apps Script 프로젝트 좌측 파일 목록에 Index.html 파일이 없어서 발생한 것입니다.

해결법 1 (추천: 10초 만에 해결)

아래의 [Code.gs 소스코드] 전체를 복사하여 Google Apps Script의 코드.gs에 붙여넣기 한 뒤, 상단 [배포] → [배포 관리] → ✏️수정 → 버전: [새 버전] → [배포]를 누르면 에러가 즉시 해결됩니다! (Index 파일이 없어도 안전하게 실행되도록 방어 코드가 적용되어 있습니다)

해결법 2 (GAS에서 직접 웹페이지 띄우기)

Apps Script 좌측 [파일 +] → [HTML]을 클릭한 뒤 이름으로 Index만 입력하고, 우측의 [전체 HTML 코드 복사] 버튼을 눌러 그 안에 붙여넣고 새 버전으로 배포하시면 됩니다.

/**
 * 도서관 좌석 및 스터디룸 예약 시스템 Google Apps Script (GAS) 백엔드
 * 구글 시트 신규 테이블 스키마 완벽 대응:
 * 1. 좌석 테이블: Seat_ID, Category, Name, Floor, Capacity, Amenities, Max_Hours, Status
 * 2. 예약 테이블: Reservation_ID, User_Name, User_Email, User_Phone, Seat_ID, Date, Start_Time, End_Time, CheckIn_Status, Status, Created_At
 */

function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const action = (e && e.parameter) ? e.parameter.action : null;
  
  if (!action) {
    try {
      return HtmlService.createHtmlOutputFromFile('Index')
        .setTitle('도서관 좌석 및 스터디룸 예약 시스템')
        .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
        .addMetaTag('viewport', 'width=device-width, initial-scale=1');
    } catch (fileErr) {
      return HtmlService.createHtmlOutput(`
        
        
        GAS API 정상 작동 중
        
        
📚 도서관 예약 API 서버 정상 작동 중
구글 시트 데이터베이스와 정상 연동되었습니다.
▶ 좌석 데이터 조회 (?action=getSeats)

      `);
    }
  }

  try {
    if (action === 'getSeats') {
      const seats = getSeatsFromSpreadsheet(ss);
      return sendJson({ success: true, data: seats });
    }
    
    if (action === 'getReservations') {
      const date = e.parameter.date;
      const allRes = getReservationsFromSpreadsheet(ss);
      const filtered = allRes.filter(r => {
        const rDate = String(r.Date || r.date || '').substring(0, 10);
        const rStatus = String(r.Status || r.status || '').toUpperCase();
        return rDate === date && rStatus !== 'CANCELLED';
      });
      return sendJson({ success: true, data: filtered });
    }
    
    if (action === 'getUserReservations') {
      const email = (e.parameter.email || '').toLowerCase();
      const allRes = getReservationsFromSpreadsheet(ss);
      const userRes = allRes.filter(r => {
        return (String(r.User_Email || r.userEmail || '')).toLowerCase() === email;
      });
      return sendJson({ success: true, data: userRes });
    }
    
    return sendJson({ success: false, message: '알 수 없는 action입니다: ' + action });
  } catch (err) {
    return sendJson({ success: false, message: err.toString() });
  }
}

function doPost(e) {
  let body;
  try {
    body = JSON.parse(e.postData.contents);
  } catch (err) {
    return sendJson({ success: false, message: 'JSON 파싱 오류: ' + err.toString() });
  }
  
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  
  try {
    if (body.action === 'createReservation') {
      const d = body.data;
      if (!d.userName || !d.userEmail || !d.seatId || !d.date || !d.startTime || !d.endTime) {
        return sendJson({ success: false, message: '필수 필드가 누락되었습니다.' });
      }
      
      const existing = getReservationsFromSpreadsheet(ss);
      const isOverlap = existing.some(r => {
        const rSeat = r.Seat_ID || r.seatId;
        const rDate = String(r.Date || r.date || '').substring(0, 10);
        const rStatus = String(r.Status || r.status || '').toUpperCase();
        if (rSeat === d.seatId && rDate === d.date && rStatus !== 'CANCELLED') {
          const s = normalizeTimeStr(d.startTime);
          const e = normalizeTimeStr(d.endTime);
          const rs = normalizeTimeStr(r.Start_Time || r.startTime);
          const re = normalizeTimeStr(r.End_Time || r.endTime);
          return (s < re && e > rs);
        }
        return false;
      });
      
      if (isOverlap) {
        return sendJson({ success: false, message: '선택하신 시간대에 이미 다른 예약이 존재합니다.' });
      }
      
      const newId = 'RES-' + Math.floor(10000000 + Math.random() * 90000000);
      const nowStr = Utilities.formatDate(new Date(), 'GMT+9', 'yyyy-MM-dd HH:mm:ss');
      
      const newRow = [
        newId,               // Reservation_ID
        d.userName,          // User_Name
        d.userEmail,         // User_Email
        d.userPhone || '',   // User_Phone
        d.seatId,            // Seat_ID
        d.date,              // Date
        d.startTime,         // Start_Time
        d.endTime,           // End_Time
        'Pending',           // CheckIn_Status (Pending)
        'Confirmed',         // Status (Confirmed)
        nowStr               // Created_At
      ];
      
      appendReservationToSpreadsheet(ss, newRow);
      
      return sendJson({
        success: true,
        message: '예약이 성공적으로 완료되었습니다.',
        reservationId: newId,
        data: {
          Reservation_ID: newId,
          User_Name: d.userName,
          User_Email: d.userEmail,
          User_Phone: d.userPhone || '',
          Seat_ID: d.seatId,
          Date: d.date,
          Start_Time: d.startTime,
          End_Time: d.endTime,
          CheckIn_Status: 'Pending',
          Status: 'Confirmed',
          Created_At: nowStr
        }
      });
    }
    
    if (body.action === 'cancelReservation') {
      const resId = body.reservationId;
      const email = (body.userEmail || '').toLowerCase();
      const updated = updateReservationField(ss, resId, email, 'Status', 'Cancelled');
      if (!updated) {
        return sendJson({ success: false, message: '예약을 찾을 수 없거나 이메일이 일치하지 않습니다.' });
      }
      return sendJson({ success: true, message: '예약이 정상적으로 취소되었습니다.' });
    }

    if (body.action === 'checkInReservation') {
      const resId = body.reservationId;
      const email = (body.userEmail || '').toLowerCase();
      const updated = updateReservationField(ss, resId, email, 'CheckIn_Status', 'CheckedIn');
      if (!updated) {
        return sendJson({ success: false, message: '예약을 찾을 수 없습니다.' });
      }
      return sendJson({ success: true, message: '입실 확인이 완료되었습니다.' });
    }
    
    return sendJson({ success: false, message: '지원하지 않는 action입니다.' });
  } catch (err) {
    return sendJson({ success: false, message: err.toString() });
  }
}

function sendJson(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function normalizeTimeStr(t) {
  if (!t) return '00:00';
  const str = String(t).trim();
  if (/^\d:\d\d/.test(str)) return '0' + str;
  return str;
}

// 좌석 데이터 추출 (단일 시트 및 분리 시트 모두 자동 지원)
function getSeatsFromSpreadsheet(ss) {
  let sheet = ss.getSheetByName('Seats') || ss.getSheetByName('좌석');
  if (!sheet) {
    sheet = ss.getSheets()[0];
  }
  const values = sheet.getDataRange().getValues();
  if (values.length === 0) return [];

  let headerRowIdx = -1;
  for (let i = 0; i < values.length; i++) {
    if (String(values[i][0]).trim().toLowerCase() === 'seat_id') {
      headerRowIdx = i;
      break;
    }
  }
  if (headerRowIdx === -1) return [];

  const headers = values[headerRowIdx].map(h => String(h).trim());
  const seats = [];

  for (let i = headerRowIdx + 1; i < values.length; i++) {
    const row = values[i];
    const id = String(row[0]).trim();
    if (!id) continue;
    if (id.toLowerCase() === 'reservation_id') break;
    if (id.toLowerCase().startsWith('string') || id.includes('식별자')) continue;

    const item = {};
    for (let j = 0; j < headers.length; j++) {
      if (headers[j]) item[headers[j]] = row[j];
    }
    seats.push(item);
  }
  return seats;
}

// 예약 데이터 추출
function getReservationsFromSpreadsheet(ss) {
  let sheet = ss.getSheetByName('Reservations') || ss.getSheetByName('예약');
  if (!sheet) {
    sheet = ss.getSheets()[0];
  }
  const values = sheet.getDataRange().getValues();
  if (values.length === 0) return [];

  let headerRowIdx = -1;
  for (let i = 0; i < values.length; i++) {
    if (String(values[i][0]).trim().toLowerCase() === 'reservation_id') {
      headerRowIdx = i;
      break;
    }
  }
  if (headerRowIdx === -1) return [];

  const headers = values[headerRowIdx].map(h => String(h).trim());
  const list = [];

  for (let i = headerRowIdx + 1; i < values.length; i++) {
    const row = values[i];
    const id = String(row[0]).trim();
    if (!id || id.toLowerCase().startsWith('string') || id.includes('고유 번호')) continue;

    const item = {};
    for (let j = 0; j < headers.length; j++) {
      if (headers[j]) item[headers[j]] = row[j];
    }
    list.push(item);
  }
  return list;
}

// 신규 예약 추가
function appendReservationToSpreadsheet(ss, newRow) {
  let sheet = ss.getSheetByName('Reservations') || ss.getSheetByName('예약');
  if (sheet) {
    sheet.appendRow(newRow);
    return;
  }
  sheet = ss.getSheets()[0];
  sheet.appendRow(newRow);
}

// 필드 업데이트
function updateReservationField(ss, resId, email, targetField, targetValue) {
  let sheet = ss.getSheetByName('Reservations') || ss.getSheetByName('예약');
  if (!sheet) sheet = ss.getSheets()[0];

  const values = sheet.getDataRange().getValues();
  let headerRowIdx = -1;
  for (let i = 0; i < values.length; i++) {
    if (String(values[i][0]).trim().toLowerCase() === 'reservation_id') {
      headerRowIdx = i;
      break;
    }
  }
  if (headerRowIdx === -1) return false;

  const headers = values[headerRowIdx].map(h => String(h).trim().toLowerCase());
  const idCol = headers.indexOf('reservation_id');
  const emailCol = headers.indexOf('user_email');
  const targetCol = headers.indexOf(targetField.toLowerCase());

  if (targetCol === -1) return false;

  for (let i = headerRowIdx + 1; i < values.length; i++) {
    const rowId = String(values[i][idCol]).trim();
    const rowEmail = String(values[i][emailCol]).trim().toLowerCase();
    if (rowId === resId && (rowEmail === email || !email)) {
      sheet.getRange(i + 1, targetCol + 1).setValue(targetValue);
      return true;
    }
  }
  return false;
}
